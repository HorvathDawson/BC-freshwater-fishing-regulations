Layer registry: the data this design implies

// @app/core/layers.ts — one entry per layer; the menu, the rail mark and mapModes all read this.
export type LayerKind = "base" | "colour-mode" | "overlay" | "setting";
export type ViewId = "map" | "water" | "conditions" | "search";
export type LayerGroup = "ground" | "lakes" | "streams" | "land" | "display";

/** Why an entry cannot be used right now. Words live in @app/core LAYER_WORDS[reason]. */
export type Availability =
  | { state: "ok" }
  | { state: "no-data";     reason: "noLakeStocking" | "noLakeSurvey" }  // prepared, not fed
  | { state: "online-only"; reason: "satelliteOffline" }                  // only when offline
  | { state: "coming-soon"; reason: "comingPools" | "comingCorridors" | "comingRelief" };

export interface LayerEntry {
  id: string;
  group: LayerGroup;
  kind: LayerKind;
  /** entries with the same `set` are one radio set (base, colour-mode); overlays are switches */
  set?: "ground" | "lakeColour" | "palette";
  title: string; sub?: string;              // words in @app/core, never in the component
  views: readonly ViewId[];                 // hidden (not greyed) on any other view
  rank: number;                             // the front page = first 3 relevant sets/entries by rank
  default: boolean;
  /** availability is derived from facts, never set by hand at run time */
  needs?: "online" | `release:${string}` | "coming-soon";
  reason?: string;                          // a LAYER_WORDS key, said when not "ok"
  /** the ONLY thing a choice may do: inputs to mapModes (one colouring pathway, colouring.ts) */
  drives: { basemap?: "map" | "imagery"; lakeMode?: string; groups?: readonly string[]; theme?: string };
  swatch?: readonly [string, string, string];  // token names, e.g. "color.stock.season"
  legend?: string;                              // a legend id, drawn from style tokens
}

/** Not entries: always on by owner rule (safety). Named in the menu, never switchable. */
export const ALWAYS_ON = ["protected", "access", "admin"] as const;

/** One resolver. The menu greys or hides from this; the rail mark counts non-defaults from it. */
export function availability(l: LayerEntry, f: { online: boolean; release: ReadonlySet<string> }): Availability {
  if (!l.needs) return { state: "ok" };
  if (l.needs === "coming-soon") return { state: "coming-soon", reason: l.reason as never };
  if (l.needs === "online") return f.online ? { state: "ok" } : { state: "online-only", reason: l.reason as never };
  return f.release.has(l.needs.slice(8)) ? { state: "ok" } : { state: "no-data", reason: l.reason as never };
}

How the sheet reads it

  • Relevant = views includes the current view. Anything else is hidden, not greyed out.
  • Front page = the first three relevant sets or overlays by rank. Everything else goes to More layers, grouped by group.
  • Control follows kind: a set of base or colour-mode entries draws as tiles; an overlay as a switch; a setting sits under Whole app.
  • Availability comes from one resolver: ok draws normally, no-data and online-only draw disabled with their reason words, and coming-soon draws a Coming chip.
  • Rail mark appears when any relevant entry differs from its default.
  • drives only sets mapModes inputs (basemap, lake mode, groups). There is no second colouring path.
  • Protected, access and admin layers are not entries. They are named once as always shown.
