Base, colour, overlays

The redesigned Layers sheet at 390 px, in the app's own tokens (themes/light.json, themes/dark.json) and faces (Bricolage Grotesque, Archivo, JetBrains Mono). It follows the Google, Gaia and onX order: pick the ground, pick what colours the water, then switch extras on and off. Every row comes from one registry entry, and every entry only feeds the inputs mapModes reads, so there is still one colouring pathway (colouring.ts).

R

The layer registry this implies

One entry per row. The sheet is a function of (registry, view, release, online): it groups entries, filters them by view, works out each one's availability, and draws it. Words live in @app/core; colours are token names.

// LAYER REGISTRY (sketch). One entry is one row in the sheet; adding a layer means adding one entry.
// Words live in @app/core (LAYER_WORDS[id]); colours are token NAMES; counts come from source.counts().
// An entry only feeds what mapModes() already reads (basemap, a layer's colour mode, active groups,
// runtime layers), so colouring.ts stays the one colouring pathway. No values are baked into tiles.

type ViewId = "map" | "conditions" | "water" | "search";      // the browse maps that open the sheet
type Kind = "base" | "colour-mode" | "overlay" | "setting";
type Group = "base" | "streams" | "lakes" | "overlays" | "app";

type Availability =                                           // WORKED OUT, never typed in (see availability())
  | { state: "ok" }
  | { state: "no-data";     reason: "noData" }                // prepared; this release carries no values
  | { state: "online-only"; reason: "offline" }               // needs the network and there is none
  | { state: "coming-soon"; reason: "comingSoon" };           // not built; NOT listed in release builds

interface LayerEntry {
  id: string;
  group: Group;
  kind: Kind;
  /** Where the row appears. A view that colours the map itself (conditions) never lists colour-mode entries. */
  views: readonly ViewId[];
  /** What choosing it does: always one of mapModes' inputs. */
  drives:
    | { basemap: "map" | "satellite" | "corridors" }          // base: ONE per sheet (radio)
    | { layer: "stream" | "lake"; mode: string; group?: string } // colour-mode: ONE per layer (radio)
    | { group: string }                                       // overlay: a $toggleable_groups group
    | { runtime: "gauges" | "spots" | "pools"; show?: string } // overlay drawn at run time
    | { theme: true };                                        // setting
  default: boolean;
  /** What it needs before it can be chosen. Missing -> no-data / online-only / coming-soon. */
  needs?: { release?: string; network?: true; built?: false };
  /** Swatch tokens for the tile or glyph, and the key it shows under the colour block when chosen. */
  swatch?: readonly string[];
  legend?: "status" | "stock" | "survey" | "flow";
  /** source.counts() key for the row's note. */
  count?: string;
}

/** The one rule that greys a row out, so nothing turns itself on or off by hand. */
export function availability(e: LayerEntry, ctx: { release: ReadonlySet<string>; online: boolean }): Availability {
  if (e.needs?.built === false) return { state: "coming-soon", reason: "comingSoon" };
  if (e.needs?.release && !ctx.release.has(e.needs.release)) return { state: "no-data", reason: "noData" };
  if (e.needs?.network && !ctx.online) return { state: "online-only", reason: "offline" };
  return { state: "ok" };
}

// The sheet's view model: group -> rows, filtered by view, coming-soon dropped in release builds.
// A choice that is unavailable is KEPT as the reader's choice and the fallback is drawn
// (satellite offline -> the map shows, card says why); mapModes receives the effective value.
//
// Words (@app/core LAYER_WORDS), e.g.
//   noData:     "No data yet"   / why: "This release carries no {what}. It switches on by itself when a release does."
//   offline:    "Offline"       / why: "Satellite needs a connection. The drawn map is showing; your choice is kept."
//   comingSoon: "Coming soon"   (development builds only)
